The actionpage.cfm

<!--- Check that form fields exist before processing --->
<cfif isDefined("form.firstName") and isDefined("form.lastName")>
  <!---
    Insert the new record into the "Employees" table.
    Replace "your_datasource_name" with the name of the DSN
    (Data Source Name) you configured in the ColdFusion Administrator.
  --->
  <cfquery name="addEmployee" datasource="your_datasource_name">
    INSERT INTO Employees (
      FirstName,
      LastName,
      Department
    )
    VALUES (
      <cfqueryparam value="#form.firstName#" cfsqltype="cf_sql_varchar">,
      <cfqueryparam value="#form.lastName#" cfsqltype="cf_sql_varchar">,
      <cfqueryparam value="#form.department#" cfsqltype="cf_sql_varchar">
    )
  </cfquery>

  <h1>Employee Added</h1>
  <p>The following employee has been added:</p>
  <cfoutput>
    <ul>
      <li>Name: #form.firstName# #form.lastName#</li>
      <li>Department: #form.department#</li>
    </ul>
  </cfoutput>
<cfelse>
  <p>No form data was submitted.</p>
</cfif>
